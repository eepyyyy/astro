The Calculation Engine

This page contains the complete jamakkol_engine.py — a single file that ties together every calculation covered in this documentation into one compute() function.

If you've read through the concept pages, every line here should make sense. If you're jumping straight here, go back and read at least Udhayam, Arudham, and Kavappu first.

The compute() function

This is the single entry point. You give it a date, time, and location. It gives you back a complete Jamakkol chart.

compute(dt_local, lat, lon, tz_h=5.5, alt=0) → Returns: dict with udhayam, arudham, kavappu, jamam, nitya_planets, upagrahas, panchangam
dt_local : datetime
The local date and time of the query. Do not convert to UTC yourself — the engine does it.
lat : float
Latitude of the astrologer's location (not the client's). Positive = North.
lon : float
Longitude of the astrologer's location. Positive = East.
tz_h : float = 5.5
Timezone offset from UTC in hours. India Standard Time = 5.5. Don't change this unless you're in a different timezone.
alt : float = 0
Elevation above sea level in meters. Affects sunrise accuracy. Use 0 if you don't know.

Full Source — jamakkol_engine.py

"""
jamakkol_engine.py
Complete Jamakkol Prasanam calculation engine.

Usage:
    from jamakkol_engine import compute
    from datetime import datetime

    result = compute(datetime(2026, 8, 4, 14, 30, 0), lat=13.0827, lon=80.2707)
"""
import pysweph as swe
from datetime import datetime, timedelta

# ─── One-time setup ───────────────────────────────────────────
swe.set_ephe_path("")                     # Use Moshier (no files needed)
swe.set_sid_mode(swe.SIDM_LAHIRI, 0, 0) # Lahiri Ayanamsa — always

# ─── Tamil Rasi Names ─────────────────────────────────────────
RASI_EN = [
    "Mesham", "Rishabham", "Mithunam", "Kadagam",
    "Simmam", "Kanni", "Thulaam", "Viruchigam",
    "Dhanusu", "Magaram", "Kumbam", "Meenam",
]
RASI_TA = [
    "மேஷம்", "ரிஷபம்", "மிதுனம்", "கடகம்",
    "சிம்மம்", "கன்னி", "துலாம்", "விருச்சிகம்",
    "தனுசு", "மகரம்", "கும்பம்", "மீனம்",
]
NAKSHATRAS = [
    "Ashwini", "Bharani", "Krittika", "Rohini", "Mrigashira",
    "Ardra", "Punarvasu", "Pushya", "Ashlesha", "Magha",
    "Purva Phalguni", "Uttara Phalguni", "Hasta", "Chitra",
    "Svati", "Vishakha", "Anuradha", "Jyeshtha", "Mula",
    "Purva Ashadha", "Uttara Ashadha", "Shravana",
    "Dhanishtha", "Shatabhisha", "Purva Bhadrapada",
    "Uttara Bhadrapada", "Revati",
]
JAMAM_PLANETS = ["Sun", "Mars", "Jupiter", "Mercury",
                 "Venus", "Saturn", "Rahu", "Ketu"]
DAY_START     = {0:"Sun",1:"Venus",2:"Mars",3:"Mercury",
                 4:"Jupiter",5:"Venus",6:"Saturn"}  # 0=Sun…6=Sat
MANDI_DAY     = [5,4,7,3,2,1,6]  # Kala index (0-based) by weekday
MANDI_NIGHT   = [0,1,4,2,3,6,7]


# ─── Helper: longitude → Rasi info ────────────────────────────
def dri(deg: float) -> dict:
    deg  = deg % 360.0
    ri   = int(deg // 30)
    ni   = int(deg // (360 / 27))
    pada = int((deg % (360/27)) // (360/108)) + 1
    return {
        "degree"     : round(deg, 4),
        "rasi"       : ri,
        "rasi_en"    : RASI_EN[ri],
        "rasi_ta"    : RASI_TA[ri],
        "deg_in_rasi": round(deg % 30, 4),
        "nakshatra"  : NAKSHATRAS[ni],
        "pada"       : pada,
    }


# ─── Helper: get sunrise/sunset for a date ────────────────────
def _rise_set(jd_start, lat, lon, alt=0):
    _, r = swe.rise_trans(jd_start, swe.SUN, swe.CALC_RISE, [lon, lat, alt], 0, 0)
    _, s = swe.rise_trans(jd_start, swe.SUN, swe.CALC_SET,  [lon, lat, alt], 0, 0)
    return r[0], s[0]


# ─── Main function ─────────────────────────────────────────────
def compute(dt_local: datetime, lat: float, lon: float,
            tz_h: float = 5.5, alt: float = 0) -> dict:
    """
    Compute a complete Jamakkol Prasanam chart.

    Parameters
    ----------
    dt_local : datetime  — local date/time of the query
    lat      : float     — astrologer's latitude  (positive = North)
    lon      : float     — astrologer's longitude (positive = East)
    tz_h     : float     — UTC offset in hours (5.5 for IST)
    alt      : float     — elevation in metres (improves sunrise accuracy)

    Returns
    -------
    dict with keys: udhayam, arudham, kavappu, jamam,
                    nitya_planets, upagrahas, panchangam,
                    sunrise_ist, sunset_ist, ayanamsa
    """

    # 1. Convert local time → UTC → Julian Day
    dt_utc = dt_local - timedelta(hours=tz_h)
    h = dt_utc.hour + dt_utc.minute/60 + dt_utc.second/3600
    jd = swe.julday(dt_utc.year, dt_utc.month, dt_utc.day, h, swe.GREG_CAL)

    ayanamsa = swe.get_ayanamsa_ut(jd)

    # 2. Sunrise / sunset (today, yesterday, tomorrow)
    midnight = int(jd) - 0.5
    sr,  ss  = _rise_set(midnight,   lat, lon, alt)
    psr, pss = _rise_set(midnight-1, lat, lon, alt)
    nsr, nss = _rise_set(midnight+1, lat, lon, alt)

    # 3. Determine which span the query falls in
    if sr <= jd < ss:
        span_s, span_e = sr, ss          # Daytime
    elif jd < sr:
        span_s, span_e = pss, sr         # Before today's sunrise
    else:
        span_s, span_e = ss, nsr         # After today's sunset

    is_day = (span_s == sr)

    # 4. Get all 9 planet positions (sidereal)
    flags = swe.FLG_SWIEPH | swe.FLG_SIDEREAL | swe.FLG_SPEED
    pids  = {"Sun":swe.SUN,"Moon":swe.MOON,"Mars":swe.MARS,
             "Mercury":swe.MERCURY,"Jupiter":swe.JUPITER,
             "Venus":swe.VENUS,"Saturn":swe.SATURN,"Rahu":swe.TRUE_NODE}
    nps = {}
    for name, pid in pids.items():
        res, _ = swe.calc_ut(jd, pid, flags)
        info = dri(res[0])
        info["retrograde"] = res[3] < 0
        info["speed"]      = round(res[3], 6)
        nps[name] = info

    ketu_lon = (nps["Rahu"]["degree"] + 180) % 360
    nps["Ketu"] = dri(ketu_lon)
    nps["Ketu"]["retrograde"] = True

    sun_rasi = nps["Sun"]["rasi"]

    # 5. Udhayam
    fraction   = (jd - span_s) / (span_e - span_s)
    udhayam    = dri(sun_rasi * 30 + fraction * 360)

    # 6. Arudham (5-minute rule — local time, seconds precision)
    tsec       = dt_local.minute * 60 + dt_local.second
    ard_rasi   = tsec // 300
    ard_deg    = ard_rasi * 30.0 + (tsec % 300) * (30.0 / 300)
    arudham    = dri(ard_deg)

    # 7. Kavappu
    dist       = (sun_rasi - ard_rasi) % 12 or 12
    kav_rasi   = (udhayam["rasi"] + dist - 1) % 12
    kavappu    = dri(kav_rasi * 30 + 15)

    # 8. Jamam number and planet positions
    jkw     = (dt_local.weekday() + 1) % 7   # 0=Sun...6=Sat
    jdur    = (span_e - span_s) / 8
    jnum    = max(1, min(8, int((jd - span_s) / jdur) + 1))
    start_p = DAY_START[jkw]
    lord    = JAMAM_PLANETS[(JAMAM_PLANETS.index(start_p) + jnum - 1) % 8]
    jpos    = {p: dri((sun_rasi + i) % 12 * 30)
               for i, p in enumerate(JAMAM_PLANETS)}

    # 9. Mandi & Gulika
    def _asc(jd2):
        c, a = swe.houses(jd2, lat, lon, b'P')
        return (a[0] - swe.get_ayanamsa_ut(jd2)) % 360

    mk_d  = MANDI_DAY[jkw]
    jm    = sr + mk_d * (ss - sr) / 8
    jg    = jm - (ss - sr) / 8

    # 10. Panchangam
    sl    = nps["Sun"]["degree"]
    ml    = nps["Moon"]["degree"]
    diff  = (ml - sl) % 360
    VAAR  = ["Somavar","Sevvai","Budhan","Viyazhan","Velli","Sani","Nyaniru"]

    # Helper to convert JD back to IST string
    def _ist(jd_val):
        y,m,d,h = swe.revjul(jd_val, swe.GREG_CAL)
        h += tz_h
        hr = int(h % 24); mn = int((h%1)*60); sc = int(((h%1)*60%1)*60)
        return f"{hr:02d}:{mn:02d}:{sc:02d} IST"

    return {
        "query_local"    : dt_local.isoformat(),
        "ayanamsa"       : round(ayanamsa, 4),
        "sunrise_ist"    : _ist(sr),
        "sunset_ist"     : _ist(ss),
        "udhayam"        : udhayam,
        "arudham"        : arudham,
        "kavappu"        : kavappu,
        "jamam"          : {
            "is_day"         : is_day,
            "number"         : jnum,
            "lord"           : lord,
            "duration_mins"  : round(jdur * 1440, 2),
            "planets"        : jpos,
        },
        "nitya_planets"  : nps,
        "upagrahas"      : {
            "mandi"  : dri(_asc(jm)),
            "gulika" : dri(_asc(jg)),
        },
        "panchangam"     : {
            "vaaram"     : VAAR[dt_local.weekday()],
            "thithi"     : (int(diff/12) % 15) + 1,
            "paksha"     : "Shukla" if diff < 180 else "Krishna",
            "nakshatra"  : nps["Moon"]["nakshatra"],
            "yogam_idx"  : int((sl + ml) % 360 / (360/27)),
        },
    }
